/*
Escreva uma função que receba um vetor de inteiros com n elementos e inverta a ordem
dos elementos no vetor. Para realizar a inversão, utilize obrigatoriamente uma pilha.
Observação: Não é permitido criar outro vetor auxiliar; a inversão deve ser realizada
exclusivamente com a pilha.
*/
#include <stdio.h>
#include <stdlib.h>

typedef struct{
    int *dados;
    int topo;
    int capac;
} Pilha;

void empilha_seq(Pilha *p, int n){
    p->dados[++p->topo] = n;
}

int desempilha_seq(Pilha *p){
    return p->dados[p->topo--];
}

int main(void){
    int n, *v, i;
    Pilha p;

    scanf("%d", &n);
    v = (int *) malloc(n * sizeof(int));
    if(v == NULL){
        return 1;
    }
    p.dados = (int *) malloc(n * sizeof(int));
    p.topo = -1;
    p.capac = n; 

    /* le vetor */
    for(i=0; i<n; i++){
        scanf("%d", &v[i]);
    }
    /* mostra vetor normal */
    for(i=0; i<n; i++){
        printf("%d ", v[i]);
    }
    printf("\n");

    for(i=0; i<n; i++)
        empilha_seq(&p, v[i]);
    for(i=0; i<n; i++)
        v[i] = desempilha_seq(&p);
    
    /* le vetor invertido */
    for(i=0; i<n; i++){
        printf("%d ", v[i]);
    }
    printf("\n");
    
    free(v);
    free(p.dados);
    return 0;
}
